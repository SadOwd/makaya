### Utiliser les Fonctions Utilitaires

```javascript
import {
  formaterMontant,
  calculerROI,
  calculerTempsCharge,
  convertirFCFAversUSD
} from './cahier-charges/data/index.js';

// Formater un montant
const budget = formaterMontant(69586200, 'FCFA');
// Résultat : "69 586 200 FCFA"

// Calculer le ROI
const roi = calculerROI(69586200, 81336000);
// Résultat : { annees: 0, mois: 10, formatee: "10 mois" }

// Calculer temps de charge
const temps = calculerTempsCharge(60, 80, 120);
// Résultat : { heures: 0, minutes: 24, formatee: "0h24" }

// Convertir FCFA en USD
const montantUSD = convertirFCFAversUSD(69586200);
// Résultat : 115977
```

### Exemple d'Affichage

```javascript
import { budgetData, modeleEconomiqueData } from './cahier-charges/data/index.js';

function AffichageBudget() {
  const total = budgetData.recapitulatif.totalInvestissementFCFA;
  const roi = modeleEconomiqueData.analyseRentabilite.retourInvestissement.roiSimple;
  
  return (
    <div>
      <h2>Budget Total : {total.toLocaleString()} FCFA</h2>
      <p>ROI : {roi}</p>
    </div>
  );
}
```
